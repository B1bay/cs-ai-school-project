# Computer Science Foundations in Modern AI
import math

class NeuralLayer:
    def __init__(self, weights: list[float], bias: float):
        self.weights = weights
        self.bias = bias

    def forward(self, inputs: list[float]) -> float:
        # Linear Algebra + Activation Algorithm
        z = sum(w * x for w, x in zip(self.weights, inputs)) + self.bias
        return 1.0 / (1.0 + math.exp(-z))

layer = NeuralLayer([0.85, 0.92, 0.78], bias=-0.5)
confidence = layer.forward([1.0, 0.9, 0.95])
print(f"AI Model Confidence: {confidence:.2%}")
